 13@dataclass
 14class RadarPoint:
 15    """
 16    A point in radar coordinates
 17
 18    Parameters
 19    ----------
 20    time : float
 21        Time in seconds since some reference epoch.
 22    range : float
 23        Range in meters relative to some reference orbit.
 24    """
 25    time: float
 26    range: float

 29@dataclass
 30class RadarBoundingBox:
 31    """
 32    A bounding box extent in radar coordinates.
 33
 34    Parameters
 35    ----------
 36    first : RadarPoint
 37        Radar coordinate of smallest time and range
 38    last : RadarPoint
 39        Radar coordinate of largest time and range
 40    """
 41    first: RadarPoint
 42    last: RadarPoint
 43
 44    def __post_init__(self):
 45        if self.first.time > self.last.time:
 46            raise ValueError("require first.time <= last.time")
 47        if self.first.range > self.last.range:
 48            raise ValueError("require first.range <= last.range")
 49
 50    @property
 51    def centroid(self) -> RadarPoint:
 52        """Return the point at the center of the region."""
 53        return RadarPoint((first.time + last.time) / 2,
 54            (first.range + last.range) / 2)

 57def get_raw_sub_swath_polygons(*, raw_bbox_lists, chirp_durations, orbit,
 58                               wavelength, azres, prf,
 59                               ellipsoid=isce3.core.Ellipsoid(),
 60                               convolution_mode="valid",
 61                               allowed_azimuth_gap=2):
 62    """
 63    Get sub-swath polygons in radar coordinates (native Doppler), accounting for
 64    areas gained or lost due to convolution in range & azimuth and cases of
 65    mixed-mode processing.
 66
 67    Parameters
 68    ----------
 69    raw_bbox_lists : list[list[RadarBoundingBox]]
 70        Bounding boxes of all subswaths for each raw data file/observation.
 71        Azimuth times should be specified in seconds relative to the orbit
 72        reference epoch.
 73    chirp_durations : list[float]
 74        Duration of transmit chirp for each raw data file, in seconds.
 75    orbit : isce3.core.Orbit
 76        Radar trajectory.
 77    wavelength : float
 78        Radar wavelength in meters.
 79    azres : float
 80        Intended azimuth resolution in meters.
 81    prf : float
 82        Pulse repetition frequency in Hz.
 83    ellipsoid : isce3.core.Ellipsoid, optional
 84        Ellipsoid describing shape of planet.
 85    convolution_mode : {"valid", "full", "same"}, optional
 86        How to handle boundary effects of focusing operation.  For "valid",
 87        only return regions that will be fully focused.  For "full", return
 88        regions with any nonzero data.  For "same", return regions that are
 89        at least halfway focused.
 90    allowed_azimuth_gap : int, optional
 91        Amount of time (specified in pulse intervals) allowed between files
 92        that is still considered contiguous.  If exceeded, the missing data
 93        will be considered invalid and masked according to `convolution_mode`.
 94
 95    Returns
 96    -------
 97    raw_polygon_lists : list[list[shapely.Polygon]]
 98        List of valid data regions for each file/observation specified in
 99        raw radar (x=range, y=time) coordinates (e.g., native Doppler).
100
101    Notes
102    -----
103    Azimuth times in the RadarBoundingBox and Doppler objects must all be
104    relative to the orbit reference epoch.
105    """
106    if len(raw_bbox_lists) != len(chirp_durations):
107        raise ValueError("Need a chirp length for each list of sub-swaths, "
108            f"but {len(raw_bbox_lists)=} while {len(chirp_durations)=}")
109
110    if convolution_mode not in {"valid", "full", "same"}:
111        raise ValueError(f"Invalid {convolution_mode=}")
112
113    # Copy user input before modifying it.
114    raw_bbox_lists = deepcopy(raw_bbox_lists)
115
116    # Handle range convolution effect.  Note that range delay is referenced
117    # to the _start_ of the chirp.
118    for chirp_duration, bboxes in zip(chirp_durations, raw_bbox_lists):
119        chirp_length = chirp_duration * isce3.core.speed_of_light / 2
120        for bbox in bboxes:
121            if convolution_mode == "full":
122                # Extend start
123                bbox.first.range -= chirp_length
124            elif convolution_mode == "valid":
125                # Truncate end
126                bbox.last.range -= chirp_length
127            elif convolution_mode == "same":
128                # Preserve length, allowing partial convolutions on both sides.
129                bbox.first.range -= chirp_length / 2
130                bbox.last.range -= chirp_length / 2
131            else:
132                assert False, "Invalid convolution mode"
133
134    # Azimuth aperture length depends on range, so we need to treat the four
135    # corners of the bounding box separately.
136    class SwathCorners:
137        def __init__(self, bbox: RadarBoundingBox):
138            self.near_begin = RadarPoint(bbox.first.time, bbox.first.range)
139            self.near_end = RadarPoint(bbox.last.time, bbox.first.range)
140            self.far_begin = RadarPoint(bbox.first.time, bbox.last.range)
141            self.far_end = RadarPoint(bbox.last.time, bbox.last.range)
142
143        def to_polygon(self):
144            # CCW order (though CW if viewed with the usual inverted time).
145            return shapely.Polygon([(pt.range, pt.time) for pt in
146                [self.near_begin, self.far_begin, self.far_end, self.near_end,
147                    self.near_begin]])
148
149    raw_corners_lists = [[SwathCorners(bbox) for bbox in bboxes]
150        for bboxes in raw_bbox_lists]
151
152    # Handle azimuth convolution effect.  Raw data is in native Doppler geometry
153    # by definition, so azimuth time is referenced to _center_ of aperture.
154    # Need to handle first and last time boundaries separately since these
155    # could be in separate files.  For now assume continuous across file
156    # boundaries.
157    for corners in raw_corners_lists[0]:
158        tmid = (corners.near_begin.time + corners.near_end.time) / 2.0
159        cpi_near = get_sar_duration(tmid, corners.near_begin.range,
160            orbit, ellipsoid, azres, wavelength)
161        cpi_far = get_sar_duration(tmid, corners.far_begin.range,
162            orbit, ellipsoid, azres, wavelength)
163
164        if convolution_mode == "full":
165            corners.near_begin.time -= cpi_near / 2
166            corners.far_begin.time -= cpi_far / 2
167        elif convolution_mode == "valid":
168            corners.near_begin.time += cpi_near / 2
169            corners.far_begin.time += cpi_far / 2
170        # else "same" -> no-op
171
172    for corners in raw_corners_lists[-1]:
173        tmid = (corners.near_begin.time + corners.near_end.time) / 2.0
174        cpi_near = get_sar_duration(tmid, corners.near_begin.range,
175            orbit, ellipsoid, azres, wavelength)
176        cpi_far = get_sar_duration(tmid, corners.far_begin.range,
177            orbit, ellipsoid, azres, wavelength)
178
179        if convolution_mode == "full":
180            corners.near_end.time += cpi_near / 2
181            corners.far_end.time += cpi_far / 2
182        elif convolution_mode == "valid":
183            corners.near_end.time -= cpi_near / 2
184            corners.far_end.time -= cpi_far / 2
185        # else "same" -> no-op
186
187    # One more wrinkle! Mark gaps in azimuth time between files
188    nfiles = len(raw_corners_lists)
189    for i in range(nfiles - 1):
190        # Only bother checking first subswath since they should all have
191        # the same time bounds.
192        t0 = raw_bbox_lists[i][0].last.time
193        t1 = raw_bbox_lists[i + 1][0].first.time
194        if (t1 - t0) * prf > allowed_azimuth_gap:
195            midpoint = (t1 - t0) / 2
196            for corners in raw_corners_lists[i]:
197                cpi_near = get_sar_duration(
198                    corners.near_end.time, corners.near_end.range,
199                    orbit, ellipsoid, azres, wavelength)
200                cpi_far = get_sar_duration(
201                    corners.far_end.time, corners.far_end.range,
202                    orbit, ellipsoid, azres, wavelength)
203
204                if convolution_mode == "full":
205                    # Don't extend further than midpoint of gap to avoid
206                    # overlapping swaths across files.
207                    corners.near_end.time += min(cpi_near / 2, midpoint)
208                    corners.far_end.time += min(cpi_far / 2, midpoint)
209                elif convolution_mode == "valid":
210                    corners.near_end.time -= cpi_near / 2
211                    corners.far_end.time -= cpi_far / 2
212
213            for corners in raw_corners_lists[i + 1]:
214                cpi_near = get_sar_duration(
215                    corners.near_begin.time, corners.near_begin.range,
216                    orbit, ellipsoid, azres, wavelength)
217                cpi_far = get_sar_duration(
218                    corners.far_begin.time, corners.far_begin.range,
219                    orbit, ellipsoid, azres, wavelength)
220
221                if convolution_mode == "full":
222                    # Don't extend further than midpoint of gap to avoid
223                    # overlapping swaths across files.
224                    corners.near_begin.time -= min(cpi_near / 2, midpoint)
225                    corners.far_begin.time -= min(cpi_far / 2, midpoint)
226                elif convolution_mode == "valid":
227                    corners.near_begin.time += cpi_near / 2
228                    corners.far_begin.time += cpi_far / 2
229
230    return [[corners.to_polygon() for corners in corners_list]
231        for corners_list in raw_corners_lists]

238def transform_polygons_raw2image(*, raw_polygon_lists, orbit, lookside,
239                                 native_doppler, wavelength,
240                                 image_grid_doppler=isce3.core.LUT2d(),
241                                 dem=isce3.geometry.DEMInterpolator(),
242                                 max_segment_length=5000.0,
243                                 rdr2geo_params=dict(),
244                                 geo2rdr_params=dict()):
245    """
246    Convert valid data polygon radar coordinates from raw data (native Doppler)
247    domain to focused image domain (usually zero Doppler for NISAR).
248
249    Parameters
250    ----------
251    raw_polygon_lists : list[list[shapely.Polygon]]
252        List of valid data regions for each file/observation specified in
253        raw radar (x=range, y=time) coordinates (e.g., native Doppler).
254    orbit : isce3.core.Orbit
255        Trajectory of radar antenna phase center.
256    lookside : isce3.core.LookSide
257        Side that radar observes (Left or Right)
258    native_doppler : isce3.core.LUT2d
259        Doppler centroid in Hz.  Time should be referenced to the same epoch
260        as the orbit.
261    wavelength : float
262        Wavelength associated with Doppler in meters.
263    image_grid_doppler : isce3.core.LUT2d, optional
264        Doppler (in Hz) associated with focused image grid geometry.
265        Defaults to zero-Doppler (NISAR convention).
266    dem : isce3.geometry.DEMInterpolator, optional
267        Digital elevation model.  Defaults to 0 m above WGS84 ellipsoid.
268    max_segment_length : float, optional
269        Length scale over which subswath boundary can be considered linear,
270        in meters.
271    rdr2geo_params : dict, optional
272        Parameters for rdr2geo_bracket
273    geo2rdr_params : dict, optional
274        Parameters for geo2rdr_bracket
275
276    Returns
277    -------
278    slc_polygon_lists : list[list[shapely.Polygon]]
279        List of valid data regions for each file/observation specified in
280        focused radar image (x=range, y=time) coordinates.
281    """
282    # Get a representative velocity to temporarily convert azimuth time to
283    # length units so we can segmentize polygon somewhat uniformly.
284    mid_time = np.mean([poly.centroid.y for poly in _flatten_nested_lists(
285        raw_polygon_lists)])
286    _, vel = orbit.interpolate(mid_time)
287    vs = np.linalg.norm(vel)
288
289    # For debugging, it's more intuitive if we remove the time offset.
290    t0 = min(poly.bounds[1] for poly in _flatten_nested_lists(
291        raw_polygon_lists))
292
293    # Helper function to convert polygon y-units from time (seconds) to space (meters)
294    multiply_vel = lambda poly: shapely.affinity.affine_transform(poly, (
295        1,  0, 0,
296        0, vs, 0,
297        0,  0, 1,
298        0, -vs * t0, 0))
299
300    # Helper function to convert polygon y-units from space (meters) to time (seconds)
301    divide_vel = lambda poly: shapely.affinity.affine_transform(poly, (
302        1, 0, 0,
303        0, 1 / vs, 0,
304        0, 0, 1,
305        0, t0, 0))
306
307    slc_polygon_lists = []
308    for raw_polygons in raw_polygon_lists:
309        slc_polygons = []
310        for raw_poly in raw_polygons:
311            # Add intermediate points between corners before transforming
312            # geometry.  Spacing specified in length units, so scale
313            # (length = velocity * time) and then back.
314            raw_poly = multiply_vel(raw_poly).segmentize(max_segment_length)
315            raw_poly = divide_vel(raw_poly)
316
317            # Transform from native Doppler to image grid Doppler
318            # (always zero for NISAR).
319            slc_coords = []
320            for (raw_range, raw_time) in zip(*raw_poly.boundary.coords.xy):
321                t, r = isce3.geometry.rdr2rdr(raw_time, raw_range, orbit,
322                    lookside, native_doppler, wavelength, dem,
323                    ellipsoid=dem.ellipsoid,
324                    doppler_out=image_grid_doppler,
325                    rdr2geo_params=rdr2geo_params,
326                    geo2rdr_params=geo2rdr_params)
327                slc_coords.append((r, t))
328            # Just in case rdr2rdr is not deterministic, force closed polygon.
329            slc_coords[-1] = slc_coords[0]
330            slc_polygons.append(shapely.Polygon(slc_coords))
331        slc_polygon_lists.append(slc_polygons)
332
333    return slc_polygon_lists

336def rasterize_subswath_polygons(slc_polygon_lists, slc_grid, threshold=2):
337    """
338    Convert valid data polygons to [start, stop) indices on the image grid.
339
340    Parameters
341    ----------
342    slc_polygon_lists : list[list[shapely.Polygon]]
343        List of valid data regions for each file/observation specified in
344        focused radar image (x=range, y=time) coordinates.
345    slc_grid : isce3.product.RadarGridParameters
346        Grid for focused image.
347    threshold : int
348        Number of range samples allowed between adjacent subswaths from
349        different observations that will still be considered contiguous and
350        merged into a single subswath.  If exceeded, the subswaths will be
351        bookkept independently.  (Relevant when reskew is significant.)
352
353    Returns
354    -------
355    sub_swaths : numpy.ndarray[np.uint32]
356        Array of [start, stop) valid data regions, shape = (nswath, npulse, 2)
357        where nswath is the number of valid sub-swaths and npulse is the length
358        of the focused image grid.
359    """
360
361    # The single-file case is straightforward, but with multiple files it's
362    # possible to have gaps in varying locations and widths if the PRF or
363    # pulse width changes.
364    nswaths = max(len(polygons) for polygons in slc_polygon_lists)
365    swaths = np.zeros((nswaths, slc_grid.shape[0], 2), dtype=np.uint32)
366
367    # Also the reskew from native to image grid Doppler means the boundaries
368    # between files may not be aligned with the grid axes, which may increase
369    # the apparent number of sub swaths.  We'll try to be smart and merge
370    # them where it's possible and pick the largest where it's not.  Avoid
371    # having a bunch of trivial extra subswaths at file boundaries.
372    tmp_swaths = np.zeros((nswaths, len(slc_polygon_lists), 2), dtype=np.uint32)
373
374    r0, r1 = slc_grid.slant_ranges[0], slc_grid.slant_ranges[-1]
375    dr = slc_grid.range_pixel_spacing
376
377    for itime in range(slc_grid.shape[0]):
378        # Scanline corresponding to a row of the radar image grid.
379        t = slc_grid.sensing_times[itime]
380        line = shapely.LineString([(r0, t), (r1, t)])
381        # Initialize to start=end, e.g., all subswaths empty/invalid.
382        tmp_swaths[...] = 0
383
384        for iobs, polygon_list in enumerate(slc_polygon_lists):
385            # Assume polygons within a single observation are disjoint.
386            # Label swath with list index so that "subSwath3" corresponds to
387            # the third subswath even if it's the only one with active samples
388            # (which can happen with azimuth recording gap + squint).
389            for iswath, poly in enumerate(polygon_list):
390                geom = line & poly
391                if geom.is_empty:
392                    # Leave subswath empty.
393                    continue
394                if geom.geom_type == "LineString":
395                    # Intersection is a single line.
396                    segment = geom
397                else:
398                    # Multiple line segments (e.g., due to reskew), so pick
399                    # the largest one.
400                    segment = max(geom.geoms, key = lambda seg: seg.length)
401                r = np.array(sorted(segment.coords.xy[0]))
402                tmp_swaths[iswath, iobs, :] = np.round((r - r0) / dr)
403
404        # Initialize with first observation.
405        swaths[:, itime, :] = tmp_swaths[:, 0, :]
406
407        # Loop over subsequent observations to 1) check for conflicts and
408        # resolve by picking the biggest region labeled "subSwathN" and
409        # 2) merge subswaths that are continuous but from different files
410        # (e.g., seamless transition in azimuth + reskew).
411        for iobs in range(1, len(slc_polygon_lists)):
412            for iswath in range(nswaths):
413                # previous best estimate and its size
414                i0, i1 = swaths[iswath, itime]
415                n = i1 - i0
416                # estimate for current observation.
417                j0, j1 = tmp_swaths[iswath, iobs]
418
419                # Cast to int to prevent unsigned underflow.
420                if abs(int(j0) - i1) <= threshold:
421                    # Join contiguous, previous first.
422                    swaths[iswath, itime] = (i0, j1)
423                elif abs(int(i0) - j1) <= threshold:
424                    # Join contiguous, current first.
425                    swaths[iswath, itime] = (j0, i1)
426                elif (j1 - j0) > n:
427                    # Conflict for this subswath label, current is bigger.
428                    swaths[iswath, itime] = (j0, j1)
429                else:
430                    # else pass because swath is either invalid or smaller
431                    assert (j0 == j1) or ((j1 - j0) <= n)
432
433    return swaths

436def get_focused_sub_swaths(raw_bbox_lists, chirp_durations, orbit,
437                           native_doppler, azres, grid,
438                           dem=isce3.geometry.DEMInterpolator(),
439                           image_grid_doppler=isce3.core.LUT2d(),
440                           rdr2geo_params=dict(),
441                           geo2rdr_params=dict(), max_segment_length=5000,
442                           convolution_mode="valid",
443                           allowed_azimuth_gap=2, allowed_range_gap=2):
444    """
445    Determine valid data regions of a focused image, considering transmit
446    gaps and gaps between files (in multi-observation processing).
447
448    Parameters
449    ----------
450    raw_bbox_lists : list[list[RadarBoundingBox]]
451        Bounding boxes of all subswaths for each raw data file/observation.
452        Azimuth times should be specified in seconds relative to the orbit
453        reference epoch.
454    chirp_durations : list[float]
455        Duration of transmit chirp for each raw data file.
456    orbit : isce3.core.Orbit
457        Trajectory of radar antenna phase center.
458    native_doppler : isce3.core.LUT2d
459        Doppler centroid in Hz.  Time should be referenced to the same epoch
460        as the orbit.
461    azres : float
462        Intended azimuth resolution in meters.
463    grid : isce3.product.RadarGridParameters
464        Grid for focused image.
465    dem : isce3.geometry.DEMInterpolator, optional
466        Digital elevation model.  Defaults to 0 m above WGS84 ellipsoid.
467    image_grid_doppler : isce3.core.LUT2d, optional
468        Doppler (in Hz) associated with focused image grid geometry.
469        Defaults to zero-Doppler (NISAR convention).
470    rdr2geo_params : dict, optional
471        Parameters for rdr2geo_bracket
472    geo2rdr_params : dict, optional
473        Parameters for geo2rdr_bracket
474    max_segment_length : float, optional
475        Length scale over which subswath boundary can be considered linear,
476        in meters.
477    convolution_mode : {"valid", "full", "same"}, optional
478        How to handle boundary effects of focusing operation.  For "valid",
479        only return regions that will be fully focused.  For "full", return
480        regions with any nonzero data.  For "same", return regions that are
481        at least halfway focused.
482    allowed_azimuth_gap : int, optional
483        Amount of time (specified in pulse intervals) allowed between files
484        that is still considered contiguous.  If exceeded, the missing data
485        will be considered invalid and masked according to `convolution_mode`.
486    allowed_range_gap : int, optional
487        Number of range samples allowed between adjacent subswaths from
488        different observations that will still be considered contiguous and
489        merged into a single subswath.  If exceeded, the subswaths will be
490        bookkept independently.  (Relevant when reskew is significant.)
491
492    Returns
493    -------
494    sub_swaths : numpy.ndarray[np.uint32]
495        Array of [start, stop) valid data regions, shape = (nswath, npulse, 2)
496        where nswath is the number of valid sub-swaths and npulse is the length
497        of the focused image grid.
498    """
499    raw_polygons = get_raw_sub_swath_polygons(raw_bbox_lists=raw_bbox_lists,
500        chirp_durations=chirp_durations, orbit=orbit,
501        wavelength=grid.wavelength, azres=azres, prf=grid.prf,
502        ellipsoid=dem.ellipsoid, convolution_mode=convolution_mode,
503        allowed_azimuth_gap=allowed_azimuth_gap)
504
505    slc_polygons = transform_polygons_raw2image(raw_polygon_lists=raw_polygons,
506        orbit=orbit, lookside=grid.lookside, native_doppler=native_doppler,
507        wavelength=grid.wavelength, dem=dem,
508        image_grid_doppler=image_grid_doppler,
509        max_segment_length=max_segment_length,
510        rdr2geo_params=rdr2geo_params, geo2rdr_params=geo2rdr_params)
511
512    return rasterize_subswath_polygons(slc_polygons, grid,
513        threshold=allowed_range_gap)

516def fill_gaps(data, swaths, value=np.complex64(0)):
517    """
518    Fill transmit gaps in raw data with a specified value.
519
520    Parameters
521    ----------
522    data : numpy.ndarray[numpy.complex64]
523        Raw data array to modify in-place.  Trailing two dimensions have shape
524        (npulse, nsamples).  Typically two dimensional, but array may have extra
525        leading dimensions as in NISAR diagnostic mode 2 (DM2) data.
526    swaths : numpy.ndarray[numpy.uint32]
527        Array of [start, stop) valid data regions, shape = (nswath, npulse, 2)
528        where nswath is the number of valid sub-swaths and npulse is the length
529        of the raw image grid.
530    value : complex, optional
531        Desired fill value.  Defaults to 0.
532
533    Notes
534    -----
535    Transmit gaps may contain recorded loop-back calibration pulses, depending
536    on the instrument mode. These should typically be zeroed-out in the raw data
537    prior to range compression when processing science data.
538    """
539    num_pulses, num_samples = data.shape[-2:]
540    num_swaths = swaths.shape[0]
541    expected_shape = (num_swaths, num_pulses, 2)
542    if swaths.shape != expected_shape:
543        raise ValueError(f"Expected swaths shape {expected_shape} but got "
544            f"shape {swaths.shape}")
545
546    # Get range index slices for the gaps of a single pulse.
547    def get_gap_slices(pulse_swaths: np.ndarray) -> Iterator[slice]:
548        # Use simply subswaths w/ valid values prior to tx gap slice
549        pulse_swaths = np.asarray(
550            [swath for swath in pulse_swaths if swath[1] > swath[0]]
551        )
552        num_swaths = pulse_swaths.shape[0]
553        # Gap leading up to first swath.
554        yield slice(None, pulse_swaths[0, 0])
555        # Gaps between swaths.
556        for i in range(num_swaths - 1):
557            gap_start = pulse_swaths[i, 1]      # end of current swath
558            gap_end = pulse_swaths[i + 1, 0]    # start of next swath
559            yield slice(gap_start, gap_end)
560        # Gap after last swath.
561        yield slice(pulse_swaths[-1, 1], None)
562    # When PRF is constant (not dithering), gap locations are constant.
563    unique_swaths = np.unique(swaths, axis=1)
564    if unique_swaths.shape[1] == 1:
565        for gap in get_gap_slices(unique_swaths[:, 0, :]):
566            data[..., gap] = value
567        return
568
569    # Otherwise we have to loop over each pulse.
570    for ipulse in range(num_pulses):
571        for gap in get_gap_slices(swaths[:, ipulse, :]):
572            data[..., ipulse, gap] = value

207                    # overlapping swaths across files.
208                    corners.near_end.time += min(cpi_near / 2, midpoint)
209                    corners.far_end.time += min(cpi_far / 2, midpoint)
210                elif convolution_mode == "valid":
211                    corners.near_end.time -= cpi_near / 2
212                    corners.far_end.time -= cpi_far / 2

337def rasterize_subswath_polygons(slc_polygon_lists, slc_grid, threshold=2):
338    """
339    Convert valid data polygons to [start, stop) indices on the image grid.
340
341    Parameters
342    ----------
343    slc_polygon_lists : list[list[shapely.Polygon]]
344        List of valid data regions for each file/observation specified in
345        focused radar image (x=range, y=time) coordinates.
346    slc_grid : isce3.product.RadarGridParameters
347        Grid for focused image.
348    threshold : int
349        Number of range samples allowed between adjacent subswaths from
350        different observations that will still be considered contiguous and
351        merged into a single subswath.  If exceeded, the subswaths will be
352        bookkept independently.  (Relevant when reskew is significant.)
353
354    Returns
355    -------
356    sub_swaths : numpy.ndarray[np.uint32]
357        Array of [start, stop) valid data regions, shape = (nswath, npulse, 2)
358        where nswath is the number of valid sub-swaths and npulse is the length
359        of the focused image grid.
360    """
361
362    # The single-file case is straightforward, but with multiple files it's
363    # possible to have gaps in varying locations and widths if the PRF or
364    # pulse width changes.
365    nswaths = max(len(polygons) for polygons in slc_polygon_lists)
366    swaths = np.zeros((nswaths, slc_grid.shape[0], 2), dtype=np.uint32)
367
368    # Also the reskew from native to image grid Doppler means the boundaries
369    # between files may not be aligned with the grid axes, which may increase
370    # the apparent number of sub swaths.  We'll try to be smart and merge
371    # them where it's possible and pick the largest where it's not.  Avoid
372    # having a bunch of trivial extra subswaths at file boundaries.
373    tmp_swaths = np.zeros((nswaths, len(slc_polygon_lists), 2), dtype=np.uint32)
374
375    r0, r1 = slc_grid.slant_ranges[0], slc_grid.slant_ranges[-1]
376    dr = slc_grid.range_pixel_spacing
377
378    for itime in range(slc_grid.shape[0]):
379        # Scanline corresponding to a row of the radar image grid.
380        t = slc_grid.sensing_times[itime]
381        line = shapely.LineString([(r0, t), (r1, t)])
382        # Initialize to start=end, e.g., all subswaths empty/invalid.
383        tmp_swaths[...] = 0
384
385        for iobs, polygon_list in enumerate(slc_polygon_lists):
386            # Assume polygons within a single observation are disjoint.
387            # Label swath with list index so that "subSwath3" corresponds to
388            # the third subswath even if it's the only one with active samples
389            # (which can happen with azimuth recording gap + squint).
390            for iswath, poly in enumerate(polygon_list):
391                geom = line & poly
392                if geom.is_empty:
393                    # Leave subswath empty.
394                    continue
395                if geom.geom_type == "LineString":
396                    # Intersection is a single line.
397                    segment = geom
398                else:
399                    # Multiple line segments (e.g., due to reskew), so pick
400                    # the largest one.
401                    segment = max(geom.geoms, key = lambda seg: seg.length)
402                r = np.array(sorted(segment.coords.xy[0]))
403                tmp_swaths[iswath, iobs, :] = np.round((r - r0) / dr)
404
405        # Initialize with first observation.
406        swaths[:, itime, :] = tmp_swaths[:, 0, :]
407
408        # Loop over subsequent observations to 1) check for conflicts and
409        # resolve by picking the biggest region labeled "subSwathN" and
410        # 2) merge subswaths that are continuous but from different files
411        # (e.g., seamless transition in azimuth + reskew).
412        for iobs in range(1, len(slc_polygon_lists)):
413            for iswath in range(nswaths):
414                # previous best estimate and its size
415                i0, i1 = swaths[iswath, itime]
416                n = i1 - i0
417                # estimate for current observation.
418                j0, j1 = tmp_swaths[iswath, iobs]
419
420                # Cast to int to prevent unsigned underflow.
421                if abs(int(j0) - i1) <= threshold:
422                    # Join contiguous, previous first.
423                    swaths[iswath, itime] = (i0, j1)
424                elif abs(int(i0) - j1) <= threshold:
425                    # Join contiguous, current first.
426                    swaths[iswath, itime] = (j0, i1)
427                elif (j1 - j0) > n:
428                    # Conflict for this subswath label, current is bigger.
429                    swaths[iswath, itime] = (j0, j1)
430                else:
431                    # else pass because swath is either invalid or smaller
432                    assert (j0 == j1) or ((j1 - j0) <= n)
433
434    return swaths
