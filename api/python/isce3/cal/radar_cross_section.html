 25def find_quadratic_poly2d_peak(coeffs: Collection[float]) -> tuple[float, float]:
 26    r"""
 27    Find the location of the peak of a quadratic bivariate polynomial.
 28
 29    Determine the location of the peak of the quadratic bivariate polynomial
 30    :math:`z = ax^2 + by^2 + cxy + dx + ey + f` by solving for the common real root of
 31    its partial derivatives with respect to :math:`x` and :math:`y`.
 32
 33    Solves the system of equations
 34
 35    .. math::
 36
 37        \begin{cases}
 38            \frac{\partial z}{\partial x} = 2ax + cy + d = 0\\
 39            \frac{\partial z}{\partial y} = 2by + cx + e = 0
 40        \end{cases}
 41
 42    whose solution :math:`(x_p, y_p)` is
 43
 44    .. math::
 45
 46        \begin{align*}
 47            x_p &= \frac{2bd - ce}{c^2 - 4ab}\\
 48            y_p &= \frac{2ae - cd}{c^2 - 4ab}
 49        \end{align*}.
 50
 51    Parameters
 52    ----------
 53    coeffs : array_like
 54        Coefficients `[a, b, c, d, e, f]` of the quadratic bivariate polynomial
 55        expression :math:`z = ax^2 + by^2 + cxy + dx + ey + f`.
 56
 57    Returns
 58    -------
 59    xp, yp : float
 60        x- and y-coordinates of the peak.
 61
 62    Raises
 63    ------
 64    ValueError
 65        If too few or too many polynomial coefficients were provided.
 66    RuntimeError
 67        If the critical point is not a local maxima.
 68    """
 69    if len(coeffs) < 6:
 70        raise ValueError(
 71            "too few polynomial coefficients were supplied: expected 6, got"
 72            f" {len(coeffs)}"
 73        )
 74    elif len(coeffs) > 6:
 75        raise ValueError(
 76            "too many polynomial coefficients were supplied: expected 6, got"
 77            f" {len(coeffs)}"
 78        )
 79
 80    # Unpack coefficients.
 81    a, b, c, d, e, _ = coeffs
 82
 83    # Check the second partial derivatives to make sure that the function is concave so
 84    # that the critical point is a local maxima.
 85    if (a >= 0.0) or (b >= 0.0):
 86        raise RuntimeError("polynomial does not contain a peak")
 87
 88    # Solve for the peak location.
 89    denom = c ** 2 - 4.0 * a * b
 90    xp = (2.0 * b * d - e * c) / denom
 91    yp = (2.0 * a * e - c * d) / denom
 92
 93    return xp, yp

 96def estimate_peak_value(data: ArrayLike) -> float:
 97    """
 98    Estimate the peak value of a 2-D discrete-time signal by fitting a quadratic.
 99
100    Parameters
101    ----------
102    data : array_like
103        A real-valued 2-D array of samples containing a single peak-like signal that may
104        be approximated by a 2-D quadratic polynomial. The data is assumed to be
105        uniformly sampled.
106
107    Returns
108    -------
109    peak : float
110        The peak value.
111
112    Raises
113    ------
114    RuntimeError
115        If the input data does not contain a well-formed peak.
116    """
117    data = np.asanyarray(data)
118
119    if data.ndim != 2:
120        raise ValueError("input data must be a 2-D array")
121    if np.issubdtype(data.dtype, np.complexfloating):
122        raise TypeError("input data must be real-valued")
123
124    # Get the pixel coordinates of the 2-D grid that the input data lies on.
125    num_rows, num_cols = data.shape
126    xcoords = np.arange(num_cols, dtype=np.float64)
127    ycoords = np.arange(num_rows, dtype=np.float64)
128
129    # Fit a quadratic 2-D polynomial to the peak.
130    poly2d = isce3.core.fit_bivariate_polynomial(xcoords, ycoords, data, degree=2)
131
132    # Get the coefficients of the quadratic bivariate polynomial.
133    coeffs = [
134        poly2d.coeffs[0, 2],  # x^2
135        poly2d.coeffs[2, 0],  # y^2
136        poly2d.coeffs[1, 1],  # xy
137        poly2d.coeffs[0, 1],  # x
138        poly2d.coeffs[1, 0],  # y
139        poly2d.coeffs[0, 0],  # 1
140    ]
141
142    # Get the (x,y) location of the peak.
143    try:
144        xp, yp = find_quadratic_poly2d_peak(coeffs)
145    except RuntimeError as e:
146        raise RuntimeError("the input array did not contain a well-formed peak") from e
147
148    # Estimate the peak power by evaluating the polynomial expression at the peak
149    # location.
150    return poly2d.eval(yp, xp)

153def zero_crossings(x: ArrayLike, y: ArrayLike) -> np.ndarray:
154    """
155    Find the positions of zero-crossings in y(x).
156
157    If the data contained no zero-crossings, an empty array is returned.
158
159    A zero crossing is considered to have occurred only when two adjacent samples have
160    opposing signs (e.g. if the `y` array is all zeros, the data does not contain any
161    zero-crossings).
162
163    Parameters
164    ----------
165    x, y : array_like
166        1-dimensional arrays containing x and y samples.
167
168    Returns
169    -------
170    xz : numpy.ndarray
171        Output array containing x-positions of zero-crossings.
172    """
173    x = np.asanyarray(x)
174    y = np.asanyarray(y)
175
176    if (x.ndim != 1) or (y.ndim != 1):
177        raise ValueError("input arrays must be 1-dimensional")
178
179    # For each pair of adjacent samples in the input `x`, check whether a sign change
180    # occurred between them (either negative-to-nonnegative or nonnegative-to-negative).
181    # Returns a boolean array where nonzero values indicate that a sign changed occurred
182    # between the corresponding element and its right-hand neighbor in `x`. Zeros are
183    # considered nonnegative.
184    def sign_change(x):
185        return np.diff(np.signbit(x))
186
187    # Get pairs of indices of elements in `y` bounding each zero-crossing point.
188    (left_indices,) = np.nonzero(sign_change(y))
189    right_indices = left_indices + 1
190
191    # Computes the x-intercept of the line through (`x0`, `y0`) and (`x1`, `y1`).
192    def x_intercept(x0, y0, x1, y1):
193        assert np.all(y1 != y0)
194        return x0 - y0 * (x1 - x0) / (y1 - y0)
195
196    # Estimate zero-crossing points with sub-sample precision by linear interpolation
197    # between neighboring samples.
198    x0 = x[left_indices]
199    y0 = y[left_indices]
200    x1 = x[right_indices]
201    y1 = y[right_indices]
202    return x_intercept(x0, y0, x1, y1)

205def estimate_peak_width(data: ArrayLike, threshold: float) -> float:
206    """
207    Estimate the width of a peak in a 1-D discrete-time signal.
208
209    Parameters
210    ----------
211    data : array_like
212        A real-valued 1-D array of samples containing a single peak-like signal that
213        exceeds `threshold`. The data is assumed to be uniformly sampled.
214    threshold : float
215        The threshold signal value at which to measure the peak width. The signal data
216        should cross the threshold exactly twice, once at the rising edge and once at
217        the falling edge of the peak.
218
219    Returns
220    -------
221    width : float
222        The width, in samples, of the peak.
223
224    Raises
225    ------
226    RuntimeError
227        If the input data did not contain a well-formed peak, or if it contained
228        multiple peaks.
229    """
230    data = np.asanyarray(data)
231
232    if data.ndim != 1:
233        raise ValueError("input data must be a 1-D array")
234    if np.issubdtype(data.dtype, np.complexfloating):
235        raise TypeError("input data must be real-valued")
236
237    # Get the approximate locations, in samples, where the input signal data intersects
238    # with the horizontal line y=threshold.
239    coords = np.arange(len(data), dtype=np.float64)
240    edges = zero_crossings(coords, data - threshold)
241
242    # There should be exactly two such locations found (the rising edge & falling edge
243    # of the peak).
244    if len(edges) < 2:
245        raise RuntimeError("the input array did not contain a well-formed peak")
246    elif len(edges) > 2:
247        raise RuntimeError("the input array contained multiple peaks")
248
249    # Compute the inner width between the two edges, in samples.
250    return edges[1] - edges[0]

253def measure_target_rcs(
254    target_llh: isce3.core.LLH,
255    img_data: ArrayLike,
256    radar_grid: isce3.product.RadarGridParameters,
257    orbit: isce3.core.Orbit,
258    doppler: isce3.core.LUT2d,
259    ellipsoid: isce3.core.Ellipsoid = isce3.core.WGS84_ELLIPSOID,
260    *,
261    nchip: int = 64,
262    upsample_factor: int = 32,
263    peak_find_domain: str = "time",
264    nfit: int = 5,
265    power_method: str = "box",
266    pthresh: float = 3.0,
267    geo2rdr_params: Optional[Mapping[str, float]] = None,
268) -> float:
269    r"""
270    Estimate a point-like target's radar cross-section (RCS) using the provided echo
271    data.
272
273    Parameters
274    ----------
275    target_llh : isce3.core.LLH
276        The target position expressed as longitude, latitude, and height above the
277        reference ellipsoid in radians, radians, and meters respectively.
278    img_data : array_like
279        The input radar domain (azimuth time by slant range) image data. A 2-D array
280        with shape (num_lines, num_range_bins). The image should be focused and
281        normalized such that its intensity represents :math:`\beta_0` values\ [1]_. The
282        data is assumed to be uniformly-sampled.
283    radar_grid : isce3.product.RadarGridParameters
284        The radar coordinates of the grid on which `img_data` is sampled.
285    orbit : isce3.core.Orbit
286        The trajectory of the radar antenna phase center.
287    doppler : isce3.core.LUT2d
288        The Doppler centroid, in hertz, of the image grid of the focused data, expressed
289        as a function of azimuth time, in seconds relative to the epoch of `radar_grid`,
290        and slant range, in meters. Note that this should be the Doppler associated with
291        the image grid, which may in general be different from the native Doppler of the
292        acquired echo data.
293    ellipsoid : isce3.core.Ellipsoid, optional
294        The geodetic reference ellipsoid, with dimensions in meters. Defaults to the
295        WGS 84 ellipsoid.
296    nchip : int, optional
297        The width, in pixels, of the square block of image data to extract centered
298        around the target position for oversampling and peak finding. Must be >= 1.
299        Defaults to 64.
300    upsample_factor : int, optional
301        The upsampling ratio. Must be >= 1. Defaults to 32.
302    peak_find_domain : {'time', 'freq'}, optional
303        Option controlling how the target peak position is estimated.
304
305        'time':
306          The default mode. The peak location is found in the time domain by detecting
307          the maximum value within a square block of image data around the expected
308          target location. The signal data is upsampled to improve precision.
309
310        'freq':
311          The peak location is found by estimating the phase ramp in the frequency
312          domain. This mode is useful when target is well-focused, has high SNR, and is
313          the only target in the neighborhood (often the case in point target
314          simulations).
315    nfit : int, optional
316        The width, in *oversampled* pixels, of the square sub-block of image data
317        (centered around the target position) to extract for fitting a quadratic
318        polynomial to the peak. Note that this is the size in pixels *after upsampling*.
319        Must be >= 3. Defaults to 5.
320    power_method : {'box', 'integrated'}, optional
321        The method for estimating the target signal power.
322
323        'box':
324          The default mode. Measures power using the rectangular box method, which
325          assumes that the target response can be approximated by a 2-D rectangular
326          function. The total power is estimated by multiplying the peak power by the
327          3dB response widths in along-track and cross-track directions.
328
329        'integrated':
330          Measures power using the integrated power method. The total power is measured
331          by summing the power of bins whose power exceeds a predefined minimum power
332          threshold.
333    pthresh : float, optional
334        The minimum power threshold, measured in dB below the peak power, for estimating
335        the target signal power using the integrated power method. This parameter is
336        ignored if `power_method` is not 'integrated'. Defaults to 3.
337    geo2rdr_params : dict or None, optional
338        An optional dict of parameters configuring the behavior of the root-finding
339        routine used in geo2rdr. The following keys are supported:
340
341        'tol_aztime':
342          Azimuth time convergence tolerance in seconds.
343
344        'time_start':
345          Start of search interval, in seconds. Defaults to ``orbit.start_time``.
346
347        'time_end':
348          End of search interval, in seconds. Defaults to ``orbit.end_time``.
349
350    Returns
351    -------
352    sigma : float
353        The measured radar cross-section, in meters squared (linear scale -- not dB).
354
355    Raises
356    ------
357    RuntimeError
358        If the target position was outside of the image grid or too near the border.
359    RuntimeError
360        If the data block surrounding the target position did not contain a well-formed
361        peak.
362
363    References
364    ----------
365    .. [1] R. K. Raney, T. Freeman, R. W. Hawkins, and R. Bamler, “A plea for radar
366       brightness,” Proceedings of IGARSS '94 - 1994 IEEE International Geoscience and
367       Remote Sensing Symposium.
368    """
369    if np.shape(img_data) != (radar_grid.length, radar_grid.width):
370        raise ValueError(
371            "shape mismatch: img_data and radar_grid must have compatible shapes"
372        )
373
374    # Ensure that time tags of orbit & radar grid are w.r.t the same epoch.
375    if orbit.reference_epoch != radar_grid.ref_epoch:
376        warnings.warn(
377            "input orbit and radar_grid are referenced to different UTC epochs -- a"
378            " copy of the orbit data will be created with updated time tags",
379            RuntimeWarning,
380        )
381        orbit = orbit.copy()
382        orbit.update_reference_epoch(radar_grid.ref_epoch)
383
384    if nchip < 1:
385        raise ValueError("nchip must be >= 1")
386    if upsample_factor < 1:
387        raise ValueError("upsample_factor must be >= 1")
388    if nfit < 3:
389        raise ValueError("nfit must be >= 3 in order to fit a quadratic polynomial")
390
391    if upsample_factor < 2:
392        warnings.warn(
393            "upsample_factor should be >= 2 to ensure that power-detecting the signal"
394            " data does not cause aliasing and that there are sufficient samples to fit"
395            " to the signal peak",
396            RuntimeWarning,
397        )
398
399    # Convert LLH object to an array containing [lon, lat, height].
400    target_llh = target_llh.to_vec3()
401
402    # Get target (x,y,z) position in ECEF coordinates.
403    target_xyz = ellipsoid.lon_lat_to_xyz(target_llh)
404
405    if geo2rdr_params is None:
406        geo2rdr_params = {}
407
408    wavelength = radar_grid.wavelength
409    look_side = radar_grid.lookside
410
411    # Run geo2rdr to get the target position in radar coordinates.
412    aztime, srange = isce3.geometry.geo2rdr_bracket(
413        xyz=target_xyz,
414        orbit=orbit,
415        doppler=doppler,
416        wavelength=wavelength,
417        side=look_side,
418        **geo2rdr_params,
419    )
420
421    # Check if target position was outside of the image grid.
422    out_of_bounds = (
423        (aztime < radar_grid.sensing_start)
424        or (aztime > radar_grid.sensing_stop)
425        or (srange < radar_grid.starting_range)
426        or (srange > radar_grid.end_range)
427    )
428    if out_of_bounds:
429        raise RuntimeError(
430            "target position in radar coordinates was outside of the supplied image"
431            " grid"
432        )
433
434    # Convert target position (aztime,srange) coordinates to (row,col) pixel
435    # coordinates within the radar grid.
436    i = (aztime - radar_grid.sensing_start) * radar_grid.prf
437    j = (srange - radar_grid.starting_range) / radar_grid.range_pixel_spacing
438
439    # Raise an exception if the target position was too near the border of the image,
440    # with insufficient margin to extract a "chip" around it.
441    near_border = (
442        (i < nchip // 2)
443        or (i > radar_grid.length - nchip // 2)
444        or (j < nchip // 2)
445        or (j > radar_grid.width - nchip // 2)
446    )
447    if near_border:
448        raise RuntimeError(
449            "target is too close to image border -- consider reducing nchip"
450        )
451
452    # Extract a small square block of image data centered around the expected target
453    # location.
454    _, _, chip = point_target_info.get_chip(img_data, i, j, nchip=nchip)
455
456    # Upsample.
457    chip_ups = point_target_info.oversample(chip, nov=upsample_factor)
458
459    # Get chip power in linear and dB.
460    chip_ups_pwr = np.abs(chip_ups) ** 2
461    chip_ups_pwr_db = pow2db(chip_ups_pwr)
462
463    # Estimate the peak position within the upsampled chip, in pixel coordinates.
464    if peak_find_domain == "time":
465        ichip, jchip = np.unravel_index(np.argmax(chip_ups_pwr), chip_ups.shape)
466    elif peak_find_domain == "freq":
467        jchip, ichip = point_target_info.measure_location_from_spectrum(chip_ups)
468    else:
469        raise ValueError(
470            f"peak_find_domain must be 'time' or 'freq', got {peak_find_domain!r}"
471        )
472
473    # Estimate the peak power of the process, in dB, by fitting a quadratic polynomial
474    # to a small subset of the upsampled chip. Note that if the target response is
475    # approximately Gaussian-shaped in linear scale, it will approximate a quadratic
476    # function in dB.
477    _, _, small_chip_pwr_db = point_target_info.get_chip(
478        chip_ups_pwr_db, ichip, jchip, nchip=nfit
479    )
480    peak_pwr_db = estimate_peak_value(small_chip_pwr_db)
481
482    # Convert dB back to linear units.
483    peak_pwr = db2pow(peak_pwr_db)
484
485    # Estimate total target signal power using rectangular box method or integrated
486    # power method.
487    if power_method == "box":
488        # Estimate target response width in azimuth bins.
489        az_cut = chip_ups_pwr_db[:, int(round(jchip))]
490        az_response_width = estimate_peak_width(az_cut, peak_pwr_db - 3.0)
491
492        # Estimate target response width in range bins.
493        rg_cut = chip_ups_pwr_db[int(round(ichip)), :]
494        rg_response_width = estimate_peak_width(rg_cut, peak_pwr_db - 3.0)
495
496        # Get total power assuming a rectangular-shaped response function.
497        target_pwr = peak_pwr * az_response_width * rg_response_width
498    elif power_method == "integrated":
499        # Get a mask of pixels above the specified threshold.
500        mask = chip_ups_pwr_db >= (peak_pwr_db - pthresh)
501
502        # Get the total power of pixels above the threshold.
503        target_pwr = np.sum(chip_ups_pwr[mask])
504    else:
505        raise ValueError(
506            f"power_method must be 'box' or 'integrated', got {power_method!r}"
507        )
508
509    # Get the approximate ground track speed of the platform, assuming that the platform
510    # trajectory can be locally approximated by a circular geocentric orbit.
511    platform_pos, platform_vel = orbit.interpolate(aztime)
512    target_pos = ellipsoid.lon_lat_to_xyz(target_llh)
513    platform_ground_speed = norm(platform_vel) * (norm(target_pos) / norm(platform_pos))
514
515    # Get the approximate azimuth pixel spacing (of the original -- not oversampled --
516    # pixels), in meters, at the target location.
517    az_pixel_spacing = platform_ground_speed / radar_grid.prf
518
519    # Convert from pixels^2 to meters^2.
520    pixel_area = az_pixel_spacing * radar_grid.range_pixel_spacing
521    ups_pixel_area = pixel_area / upsample_factor ** 2
522    return target_pwr * ups_pixel_area

$$\begin{cases} \frac{\partial z}{\partial x} = 2ax + cy + d = 0\ \frac{\partial z}{\partial y} = 2by + cx + e = 0 \end{cases}$$

whose solution \( (x_p, y_p) \) is

$$\begin{align} x_p &= \frac{2bd - ce}{c^2 - 4ab}\ y_p &= \frac{2ae - cd}{c^2 - 4ab} \end{align}.$$

Parameters

coeffs : array_like Coefficients [a, b, c, d, e, f] of the quadratic bivariate polynomial expression \( z = ax^2 + by^2 + cxy + dx + ey + f \).

Returns

xp, yp : float x- and y-coordinates of the peak.

Raises

ValueError If too few or too many polynomial coefficients were provided. RuntimeError If the critical point is not a local maxima.

154def zero_crossings(x: ArrayLike, y: ArrayLike) -> np.ndarray:
155    """
156    Find the positions of zero-crossings in y(x).
157
158    If the data contained no zero-crossings, an empty array is returned.
159
160    A zero crossing is considered to have occurred only when two adjacent samples have
161    opposing signs (e.g. if the `y` array is all zeros, the data does not contain any
162    zero-crossings).
163
164    Parameters
165    ----------
166    x, y : array_like
167        1-dimensional arrays containing x and y samples.
168
169    Returns
170    -------
171    xz : numpy.ndarray
172        Output array containing x-positions of zero-crossings.
173    """
174    x = np.asanyarray(x)
175    y = np.asanyarray(y)
176
177    if (x.ndim != 1) or (y.ndim != 1):
178        raise ValueError("input arrays must be 1-dimensional")
179
180    # For each pair of adjacent samples in the input `x`, check whether a sign change
181    # occurred between them (either negative-to-nonnegative or nonnegative-to-negative).
182    # Returns a boolean array where nonzero values indicate that a sign changed occurred
183    # between the corresponding element and its right-hand neighbor in `x`. Zeros are
184    # considered nonnegative.
185    def sign_change(x):
186        return np.diff(np.signbit(x))
187
188    # Get pairs of indices of elements in `y` bounding each zero-crossing point.
189    (left_indices,) = np.nonzero(sign_change(y))
190    right_indices = left_indices + 1
191
192    # Computes the x-intercept of the line through (`x0`, `y0`) and (`x1`, `y1`).
193    def x_intercept(x0, y0, x1, y1):
194        assert np.all(y1 != y0)
195        return x0 - y0 * (x1 - x0) / (y1 - y0)
196
197    # Estimate zero-crossing points with sub-sample precision by linear interpolation
198    # between neighboring samples.
199    x0 = x[left_indices]
200    y0 = y[left_indices]
201    x1 = x[right_indices]
202    y1 = y[right_indices]
203    return x_intercept(x0, y0, x1, y1)

207    """
208    Estimate the width of a peak in a 1-D discrete-time signal.
209
210    Parameters
211    ----------
212    data : array_like
213        A real-valued 1-D array of samples containing a single peak-like signal that
214        exceeds `threshold`. The data is assumed to be uniformly sampled.
215    threshold : float
216        The threshold signal value at which to measure the peak width. The signal data
217        should cross the threshold exactly twice, once at the rising edge and once at
218        the falling edge of the peak.
219
220    Returns
221    -------

224
225    Raises
226    ------
227    RuntimeError
228        If the input data did not contain a well-formed peak, or if it contained

230    """
231    data = np.asanyarray(data)
232
233    if data.ndim != 1:
234        raise ValueError("input data must be a 1-D array")
235    if np.issubdtype(data.dtype, np.complexfloating):
236        raise TypeError("input data must be real-valued")
237
238    # Get the approximate locations, in samples, where the input signal data intersects
239    # with the horizontal line y=threshold.
240    coords = np.arange(len(data), dtype=np.float64)
241    edges = zero_crossings(coords, data - threshold)
242
243    # There should be exactly two such locations found (the rising edge & falling edge
244    # of the peak).
245    if len(edges) < 2:

247    elif len(edges) > 2:
248        raise RuntimeError("the input array contained multiple peaks")
249
250    # Compute the inner width between the two edges, in samples.
251    return edges[1] - edges[0]
