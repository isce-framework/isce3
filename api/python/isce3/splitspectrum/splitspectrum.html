 29    def load_from_slc(cls, slc_product, freq):
 30        """Get meta data from SLC object.
 31        Parameters
 32        ----------
 33        slc_product : nisar.products.readers.SLC
 34            slc object
 35        freq : {'A', 'B'}
 36            frequency band
 37        Returns
 38        -------
 39        meta_data : BandpassMetaData
 40            bandpass meta data object
 41        """
 42        rdr_grid = slc_product.getRadarGrid(freq)
 43        rg_sample_freq = \
 44            isce3.core.speed_of_light * 0.5 /\
 45            rdr_grid.range_pixel_spacing
 46        is_close = math.isclose(rg_sample_freq,
 47                                np.round(rg_sample_freq),
 48                                rel_tol=1e-8)
 49        if is_close:
 50            rg_sample_freq = np.round(rg_sample_freq)
 51
 52        rg_bandwidth = \
 53            slc_product.getSwathMetadata(freq).processed_range_bandwidth
 54        center_frequency = \
 55            isce3.core.speed_of_light / rdr_grid.wavelength
 56        return cls(rdr_grid.range_pixel_spacing, rdr_grid.wavelength,
 57                   rg_sample_freq, rg_bandwidth, center_frequency,
 58                   rdr_grid.slant_range)

 61def check_range_bandwidth_overlap(ref_slc, sec_slc, pols):
 62    """Check if bandpass is needed.
 63
 64    If the two SLCs differ in center frequency or bandwidth, then
 65    one SLC shall be bandpassed to a common frequency band. If
 66    necessary, determine which SLC will be bandpassed
 67
 68    Parameters
 69    ----------
 70    ref_slc : nisar.products.readers.SLC
 71        Reference SLC object
 72    sec_slc : nisar.products.readers.SLC
 73        Secondary SLC object
 74    pols : dict
 75        Dict keying frequency ('A' or 'B') to list of polarization values.
 76
 77    Returns
 78    -------
 79    mode : dict
 80        Dict mapping frequency band (e.g. "A" or "B") to
 81        SLC to be bandpassed ("ref" or "sec").
 82    """
 83    mode = dict()
 84
 85    for freq, pol_list in pols.items():
 86        ref_meta_data = BandpassMetaData.load_from_slc(ref_slc, freq)
 87        sec_meta_data = BandpassMetaData.load_from_slc(sec_slc, freq)
 88
 89        ref_wvl = ref_meta_data.wavelength
 90        sec_wvl = sec_meta_data.wavelength
 91        ref_bw = ref_meta_data.rg_bandwidth
 92        sec_bw = sec_meta_data.rg_bandwidth
 93
 94        # check if two SLCs have same bandwidth and center frequency
 95        if (ref_wvl != sec_wvl) or (ref_bw != sec_bw):
 96            if ref_bw > sec_bw:
 97                mode[freq] = 'ref'
 98            else:
 99                mode[freq] = 'sec'
100    return mode

144    def bandpass_shift_spectrum(self,
145                                slc_raster,
146                                low_frequency,
147                                high_frequency,
148                                new_center_frequency,
149                                window_function,
150                                window_shape=0.25,
151                                fft_size=None,
152                                resampling=True
153                                ):
154        """Bandpass SLC for a given bandwidth and shift the bandpassed
155        spectrum to a new center frequency
156
157        Parameters
158        ----------
159        slc_raster : numpy.ndarray
160            numpy array of slc raster,
161        low_frequency : float
162            low  frequency of band to be passed [Hz]
163        high_frequency : float
164            high frequency band to be passed [Hz]
165        new_center_frequency : float
166            new center frequency for new bandpassed slc [Hz]
167        window_function : str
168            window type {tukey, kaiser, cosine}
169        window_shape : float
170            parameter for the raised cosine filter (e.g. 0 ~ 1)
171        fft_size : int
172            fft size.
173        resampling : bool
174            if True, then resample SLC and meta data with new range spacing
175            If False, return SLC and meta with original range spacing
176
177        Returns
178        -------
179        resampled_slc or slc_demodulate: numpy.ndarray
180            numpy array of bandpassed slc
181            if resampling is True,
182            return resampled slc with bandpass and demodulation
183            if resampling is False,
184            return slc with bandpass and demodulation without resampling
185        meta : dict
186            dict containing meta data of bandpassed slc
187            center_frequency, rg_bandwidth, range_spacing, slant_range
188        """
189        error_channel = journal.error(
190            'splitspectrum.bandpass_shift_spectrum')
191        slc_bp = self.bandpass_spectrum(
192            slc_raster=slc_raster,
193            low_frequency=low_frequency,
194            high_frequency=high_frequency,
195            window_function=window_function,
196            window_shape=window_shape,
197            fft_size=fft_size,
198            )
199
200        # demodulate the SLC to be baseband to new center frequency
201        # if fft_size > width, then crop the spectrum from 0 to width
202        diff_frequency = self.center_frequency - new_center_frequency
203        height, width = slc_raster.shape
204
205        slc_raster = np.asanyarray(slc_raster, dtype=np.complex128)
206
207        slc_demodulate = self.demodulate_slc(
208            slc_bp[:, :width],
209            diff_frequency,
210            self.rg_sample_freq
211        )
212
213        # update metadata with new parameters
214        new_bandwidth = high_frequency - low_frequency
215        new_rg_sample_freq = np.abs(new_bandwidth) * \
216            self.sampling_bandwidth_ratio
217        meta = {
218            'center_frequency': new_center_frequency,
219            'rg_bandwidth': new_bandwidth,
220            'rg_sample_freq': new_rg_sample_freq
221        }
222
223        # Resampling is required when the output bandpassed SLC needs to be
224        # downsampled, for example, to match frequency A with frequency B 
225        # after bandpassing frequency A. This process modifies the pixel s
226        # pacing and the corresponding slant range.
227        if resampling:
228            resampling_scale_factor = self.rg_sample_freq / new_rg_sample_freq
229
230            # convert to integer
231            # Due to floating-point precision limits, the computed
232            # resampling scale factor may not be exactly an integer.
233            if self.rg_sample_freq % new_rg_sample_freq > 1e-7:
234                err_msg = 'Resampling scaling factor ' \
235                          f'{resampling_scale_factor} must be an integer.'
236                error_channel.log(err_msg)
237                raise ValueError(err_msg)
238
239            resampling_scale_factor = np.round(resampling_scale_factor)
240
241            sub_width = int(width / resampling_scale_factor)
242
243            x_cand = np.arange(1, width + 1)
244            # find the maximum of the multiple of resampling_scale_factor
245            resample_width_end = np.max(x_cand[x_cand %
246                                               resampling_scale_factor == 0])
247
248            # resample SLC
249            resampled_slc = resample(
250                slc_demodulate[:, :resample_width_end], sub_width, axis=1)
251
252            meta['range_spacing'] = \
253                self.rg_pxl_spacing * resampling_scale_factor
254            meta['slant_range'] = \
255                self.slant_range(0) + \
256                np.arange(sub_width) * meta['range_spacing']
257
258            return resampled_slc, meta
259
260        else:
261            meta['range_spacing'] = self.rg_pxl_spacing
262            meta['slant_range'] = \
263                self.slant_range(0) + \
264                np.arange(width) * meta['range_spacing']
265
266            return slc_demodulate, meta

268    def bandpass_spectrum(self,
269                          slc_raster,
270                          low_frequency,
271                          high_frequency,
272                          window_function,
273                          window_shape=0.25,
274                          fft_size=None,
275                          ):
276        """Bandpass SLC for given center frequency and bandwidth
277
278        Parameters
279        ----------
280        slc_raster : numpy.ndarray
281            numpy array of slc raster,
282        low_frequency : float
283            low  frequency of band to be passed [Hz]
284        high_frequency : float
285            high frequency band to be passed [Hz]
286        window_function: str
287            window type {'tukey', 'kaiser', 'cosine'}
288        window_shape : float
289            parameter for the window shape
290            kaiser 0<= window_shape < inf
291            tukey and cosine 0 <= window_shape <= 1
292        fft_size : int
293            fft size.
294
295        Returns
296        -------
297        slc_bandpassed : numpy.ndarray
298            numpy array of bandpassed slc
299        """
300        error_channel = journal.error('splitspectrum.bandpass_spectrum')
301
302        height, width = slc_raster.shape
303        slc_raster = np.asanyarray(slc_raster, dtype='complex')
304
305        if high_frequency < low_frequency:
306            raise ValueError("low_frequency cannot exceed high_frequency.")
307
308        # If fft_size is not specified, use scipy's next_fast_len to determine
309        # the optimal size. This typically results in faster FFT computation
310        # than using the next power of 2, as it selects lengths that are highly
311        # composite (products of small prime factors).
312        if fft_size is None:
313            fft_size = next_fast_len(width)
314
315        if fft_size < width:
316            msg = f"Requested fft_size={fft_size} < number of range bins={width}."
317            error_channel.log(msg)
318            raise ValueError(msg)
319
320        rg_sample_freq = self.rg_sample_freq
321        rg_bandwidth = self.rg_bandwidth
322        center_frequency = self.center_frequency
323
324        padded_slc = np.zeros((height, fft_size), dtype='complex')
325        padded_slc[:, :width] = slc_raster
326
327        # construct window to be deconvolved
328        # from the original SLC in freq domain
329        window_target = self.get_range_bandpass_window(
330            center_frequency=0,
331            freq_low=-rg_bandwidth/2,
332            freq_high=rg_bandwidth/2,
333            sampling_frequency=rg_sample_freq,
334            fft_size=fft_size,
335            window_function=window_function,
336            window_shape=window_shape
337            )
338        # construct window to bandpass spectrum
339        # for given low and high frequencies
340        window_bandpass = self.get_range_bandpass_window(
341            center_frequency=0,
342            freq_low=low_frequency - center_frequency,
343            freq_high=high_frequency - center_frequency,
344            sampling_frequency=rg_sample_freq,
345            fft_size=fft_size,
346            window_function=window_function,
347            window_shape=window_shape
348            )
349
350        new_bandwidth = high_frequency - low_frequency
351        new_rg_sample_freq = self.sampling_bandwidth_ratio * new_bandwidth
352        resampling_scale_factor = rg_sample_freq / new_rg_sample_freq
353
354        # remove the windowing effect from the spectrum
355        spectrum_target = fft(padded_slc, n=fft_size, workers=-1)
356        spectrum_target = np.divide(spectrum_target,
357                                    window_target,
358                                    out=np.zeros_like(spectrum_target),
359                                    where=window_target != 0)
360
361        filtered_spectrum = (spectrum_target *
362                             window_bandpass *
363                             np.sqrt(resampling_scale_factor))
364
365        slc_bandpassed_full = ifft(filtered_spectrum, n=fft_size, workers=-1)
366
367        slc_bandpassed = slc_bandpassed_full[:, :width]
368
369        return slc_bandpassed

371    def demodulate_slc(self, slc_array, diff_frequency, rg_sample_freq):
372        """ Demodulate SLC
373
374        If diff_frequency is not zero, then the spectrum of SLC is shifted
375        so that the sub-band slc is demodulated to center the sub band spectrum
376
377        Parameters
378        ----------
379        slc_array : numpy.ndarray
380            SLC raster or block of SLC raster
381        diff_frequency : float
382            difference between original and new center frequency [Hz]
383        rg_sample_freq : float
384            range sampling frequency [Hz]
385
386        Returns
387        -------
388        slc_baseband  : numpy.ndarray
389            demodulated SLC
390        """
391        height, width = slc_array.shape
392        range_time = np.arange(width) / rg_sample_freq
393        slc_shifted = slc_array * np.exp(1j * 2.0 * np.pi
394                                         * diff_frequency * range_time)
395        return slc_shifted

417    def get_range_bandpass_window(self,
418                                  center_frequency,
419                                  sampling_frequency,
420                                  fft_size,
421                                  freq_low,
422                                  freq_high,
423                                  window_function='tukey',
424                                  window_shape=0.25):
425        '''Get range bandpass window such as Tukey, Kaiser, cosine.
426        Window is constructed in frequency domain from low to high frequencies
427        with given window_function and shape.
428
429        Parameters
430        ----------
431        center_frequency : float
432            Center frequency of frequency bin [Hz]
433            If slc is basebanded, center_frequency can be 0.
434        sampling_frequency : float
435            sampling frequency [Hz]
436        fft_size : int
437            fft size
438        freq_low : float
439            low frequency to be passed [Hz]
440        freq_high: float
441            high frequency to be passed [Hz]
442        window_function : str
443            window type {tukey, kaiser, cosine}
444        window_shape : float
445            parameter for the window shape
446            kaiser 0<= window_shape < inf
447            tukey and cosine 0 <= window_shape <= 1
448
449        Returns
450        -------
451        filter_1d : np.ndarray
452            one dimensional bandpass filter in frequency domain
453        '''
454        error_channel = journal.error(
455            'splitspectrum.get_range_bandpass_window')
456        # construct the frequency bin [Hz]
457        frequency = self.freq_spectrum(
458                    cfrequency=center_frequency,
459                    dt=1.0/sampling_frequency,
460                    fft_size=fft_size
461                    )
462
463        window_kind = window_function.lower()
464
465        # Windowing effect will appear from freq_low to freq_high
466        # for given frequency bin
467        if window_kind == 'tukey':
468            if not (0 <= window_shape <= 1):
469                err_str = f"Expected window_shape between 0 and 1, got {window_shape}."
470                error_channel.log(err_str)
471                raise ValueError(err_str)
472
473            filter_1d = self.construct_range_bandpass_tukey(
474                frequency_range=frequency,
475                freq_low=freq_low,
476                freq_high=freq_high,
477                window_shape=window_shape
478            )
479
480        elif window_kind == 'kaiser':
481            if not (window_shape > 0):
482                err_str = f"Expected pedestal bigger than 0, got {window_shape}."
483                error_channel.log(err_str)
484                raise ValueError(err_str)
485
486            filter_1d = self.construct_range_bandpass_kaiser(
487                frequency_range=frequency,
488                freq_low=freq_low,
489                freq_high=freq_high,
490                window_shape=window_shape
491            )
492
493        elif window_kind == 'cosine':
494            if not (0 <= window_shape <= 1):
495                err_str = f"Expected window_shape between 0 and 1, got {window_shape}."
496                error_channel.log(err_str)
497                raise ValueError(err_str)
498            filter_1d = self.construct_range_bandpass_cosine(
499                frequency_range=frequency,
500                freq_low=freq_low,
501                freq_high=freq_high,
502                window_shape=window_shape
503            )
504
505        else:
506            err_str = f"window {window_kind} not in (Kaiser, Cosine, Tukey)."
507            error_channel.log(err_str)
508            raise ValueError(err_str)
509
510        return filter_1d

574    def _construct_range_bandpass_kaiser_cosine(
575            self,
576            frequency_range,
577            freq_low,
578            freq_high,
579            window_function,
580            window_shape):
581        '''Generate a Kaiser or cosine bandpass window
582
583        Parameters
584        ----------
585        frequency_range : np.ndarray
586            Discrete Fourier Transform sample frequency range bins[Hz]
587        freq_low : float
588            low frequency to be passed [Hz]
589        freq_high: float
590            high frequency to be passed [Hz]
591        window_function : class function
592            window type {np.kaiser, cosine_window}
593        window_shape : float
594            parameter for the kaiser window
595
596        Returns
597        -------
598        filter_1d : np.ndarray
599            one dimensional kaiser bandpass filter in frequency domain
600        '''
601        error_channel = journal.error(
602            'splitspectrum._construct_range_bandpass_kaiser_cosine')
603
604        subbandwidth = np.abs(freq_high - freq_low)
605        fft_size = len(frequency_range)
606
607        if freq_high > np.max(frequency_range):
608            err_str = "High frequency is out of frequency bins."
609            error_channel.log(err_str)
610            raise ValueError(err_str)
611
612        if freq_low < np.min(frequency_range):
613            err_str = "Low frequency is out of frequency bins."
614            error_channel.log(err_str)
615            raise ValueError(err_str)
616
617        # sampling frequency is 1.2 times wider than bandwidth for NISAR
618        sampling_bandwidth_ratio = self.sampling_bandwidth_ratio
619
620        sampling_low_frequency = \
621            freq_low - (sampling_bandwidth_ratio - 1) * subbandwidth * 0.5
622        sampling_high_frequency = \
623            freq_high + (sampling_bandwidth_ratio - 1) * subbandwidth * 0.5
624
625        # index for low and high sampling frequency in frequency_range
626        idx_freq_low = np.abs(
627            frequency_range - sampling_low_frequency).argmin()
628        idx_freq_high = np.abs(
629            frequency_range - sampling_high_frequency).argmin()
630
631        if idx_freq_low >= idx_freq_high:
632            subband_length = idx_freq_high + fft_size - idx_freq_low + 1
633        else:
634            subband_length = idx_freq_high - idx_freq_low + 1
635
636        filter_1d = np.zeros([fft_size], dtype='complex')
637
638        # window_function is function class {np.kaiser or consine}
639        subwindow = window_function(subband_length, window_shape)
640
641        if idx_freq_low >= idx_freq_high:
642            filter_1d[idx_freq_low:] = subwindow[0:fft_size - idx_freq_low]
643            filter_1d[: idx_freq_high + 1] = subwindow[fft_size - idx_freq_low:]
644        else:
645            filter_1d[idx_freq_low:idx_freq_high+1] = subwindow
646
647        return filter_1d

649    def construct_range_bandpass_tukey(self,
650                                       frequency_range,
651                                       freq_low,
652                                       freq_high,
653                                       window_shape):
654        '''Generate a Tukey (raised-cosine) window
655
656        Parameters
657        ----------
658        frequency_range : np.ndarray
659            Discrete Fourier Transform sample frequency range [Hz]
660        freq_low : float
661            low frequency to be passed [Hz]
662        freq_high, : list of float
663            high frequency to be passed [Hz]
664        window_shape : float
665            parameter for the Tukey (raised cosine) filter
666
667        Returns
668        -------
669        filter_1d : np.ndarray
670            one dimensional Tukey bandpass filter in frequency domain
671        '''
672
673        fft_size = len(frequency_range)
674        freq_mid = 0.5 * (freq_low + freq_high)
675        subbandwidth = np.abs(freq_high - freq_low)
676        df = 0.5 * subbandwidth * window_shape
677
678        filter_1d = np.zeros(fft_size, dtype='complex')
679        for i in range(0, fft_size):
680            # Get the absolute value of shifted frequency
681            freq = frequency_range[i]
682            freqabs = np.abs(freq - freq_mid)
683            # Passband. i.e. range of frequencies that can pass
684            # through a filter
685            if (freq <= (freq_high - df)) and (freq >= (freq_low + df)):
686                filter_1d[i] = 1

307            raise ValueError("low_frequency cannot exceed high_frequency.")
308

311        # than using the next power of 2, as it selects lengths that are highly
312        # composite (products of small prime factors).
313        if fft_size is None:
314            fft_size = next_fast_len(width)

316        if fft_size < width:
317            msg = f"Requested fft_size={fft_size} < number of range bins={width}."
318            error_channel.log(msg)
319            raise ValueError(msg)
320
321        rg_sample_freq = self.rg_sample_freq
322        rg_bandwidth = self.rg_bandwidth
323        center_frequency = self.center_frequency
324
325        padded_slc = np.zeros((height, fft_size), dtype='complex')
326        padded_slc[:, :width] = slc_raster

330        window_target = self.get_range_bandpass_window(
331            center_frequency=0,

333            freq_high=rg_bandwidth/2,
334            sampling_frequency=rg_sample_freq,

336            window_function=window_function,
337            window_shape=window_shape

342            center_frequency=0,

344            freq_high=high_frequency - center_frequency,
345            sampling_frequency=rg_sample_freq,
346            fft_size=fft_size,
347            window_function=window_function,
348            window_shape=window_shape
349            )

352        new_rg_sample_freq = self.sampling_bandwidth_ratio * new_bandwidth
353        resampling_scale_factor = rg_sample_freq / new_rg_sample_freq
354

356        spectrum_target = fft(padded_slc, n=fft_size, workers=-1)
357        spectrum_target = np.divide(spectrum_target,
358                                    window_target,
359                                    out=np.zeros_like(spectrum_target),
360                                    where=window_target != 0)
361

363                             window_bandpass *
364                             np.sqrt(resampling_scale_factor))
365
366        slc_bandpassed_full = ifft(filtered_spectrum, n=fft_size, workers=-1)
367
368        slc_bandpassed = slc_bandpassed_full[:, :width]
369
370        return slc_bandpassed

583
584        Parameters
585        ----------
586        frequency_range : np.ndarray
587            Discrete Fourier Transform sample frequency range bins[Hz]
588        freq_low : float
589            low frequency to be passed [Hz]
590        freq_high: float
591            high frequency to be passed [Hz]
592        window_function : class function
593            window type {np.kaiser, cosine_window}
594        window_shape : float
595            parameter for the kaiser window
596
597        Returns
598        -------
599        filter_1d : np.ndarray
600            one dimensional kaiser bandpass filter in frequency domain
601        '''
602        error_channel = journal.error(
603            'splitspectrum._construct_range_bandpass_kaiser_cosine')
604
605        subbandwidth = np.abs(freq_high - freq_low)
606        fft_size = len(frequency_range)
607
608        if freq_high > np.max(frequency_range):
609            err_str = "High frequency is out of frequency bins."
610            error_channel.log(err_str)
611            raise ValueError(err_str)
612
613        if freq_low < np.min(frequency_range):
614            err_str = "Low frequency is out of frequency bins."
615            error_channel.log(err_str)
616            raise ValueError(err_str)
617
618        # sampling frequency is 1.2 times wider than bandwidth for NISAR
619        sampling_bandwidth_ratio = self.sampling_bandwidth_ratio
620
621        sampling_low_frequency = \
622            freq_low - (sampling_bandwidth_ratio - 1) * subbandwidth * 0.5
623        sampling_high_frequency = \
624            freq_high + (sampling_bandwidth_ratio - 1) * subbandwidth * 0.5
625
626        # index for low and high sampling frequency in frequency_range
627        idx_freq_low = np.abs(
628            frequency_range - sampling_low_frequency).argmin()
629        idx_freq_high = np.abs(
630            frequency_range - sampling_high_frequency).argmin()
631
632        if idx_freq_low >= idx_freq_high:
633            subband_length = idx_freq_high + fft_size - idx_freq_low + 1
634        else:
635            subband_length = idx_freq_high - idx_freq_low + 1
636
637        filter_1d = np.zeros([fft_size], dtype='complex')
638
639        # window_function is function class {np.kaiser or consine}

657        Parameters
